### outbound-url-guard.ts

```typescript
import { lookup } from "node:dns/promises";
import { isIP } from "node:net";

const BLOCKED_V4 = [
  /^127\./,           // loopback
  /^10\./,            // private A
  /^169\.254\./,      // link-local / AWS metadata
  /^192\.168\./,      // private C
  /^172\.(1[6-9]|2\d|3[01])\./,  // private B
  /^0\./,             // this network
];

export async function assertPublicHttpUrl(raw: string): Promise<URL> {
  const url = new URL(raw);
  if (url.protocol !== "https:" && url.protocol !== "http:") {
    throw new Error(`Scheme '${url.protocol}' is not allowed`);
  }
  const host = url.hostname;
  const addrs = isIP(host)
    ? [{ address: host }]
    : await lookup(host, { all: true });
  for (const { address } of addrs) {
    const v = isIP(address);
    if (v === 4 && BLOCKED_V4.some((re) => re.test(address))) {
      throw new Error(`Refusing connection to private/loopback address ${address}`);
    }
    if (v === 6) {
      const lower = address.toLowerCase();
      if (
        lower === "::1" ||
        lower.startsWith("fc") ||
        lower.startsWith("fd") ||
        lower.startsWith("fe80")
      ) {
        throw new Error(`Refusing connection to private/loopback IPv6 address ${address}`);
      }
    }
  }
  return url;
}
```

### metrics/route.ts patch

```typescript
import { timingSafeEqual } from "node:crypto";
import { env } from "@/shared/config/env";

export async function GET(request: Request) {
  if (env.METRICS_TOKEN) {
    const auth = request.headers.get("authorization") ?? "";
    const provided = auth.startsWith("Bearer ") ? auth.slice(7) : "";
    const expected = env.METRICS_TOKEN;
    const ok =
      provided.length === expected.length &&
      timingSafeEqual(Buffer.from(provided), Buffer.from(expected));
    if (!ok) {
      return new Response("Unauthorized", { status: 401 });
    }
  }
  // ... existing prometheus output
}
```
